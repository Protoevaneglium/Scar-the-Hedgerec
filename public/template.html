2. Sculpt Terrain Piece-by-Piece (`buildTerrain(b, terrain)`)
buildTerrain(b, terrain) {
    const main = b.path(0, 580)
        .flat(480)                  // Flat start runway
        .bump(420, 28, +1)          // Smooth 28° hill
        .ease(120, 0, 0.35)         // Ease into downhill slope
        .line(320, 0.35)            // Straight downhill ramp
        .ease(120, 0.35, 0)         // Level out smoothly
        .flat(1400);                // Straightaway
    b.addFloor(main, { name: 'main floor' });

    // Elevated jump-through platform shelf + 360° Loop:
    b.addShelf(b.path(700, 420).flat(360), { tier: 'high' });
    b.addLoop(1900, 692 - 128, 128, 'Loop #1');
}
